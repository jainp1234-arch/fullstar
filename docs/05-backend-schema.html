1Rules for Claude Code

  1. Create the database exactly from the SQL in this document, split into the Flyway files listed in section 14. Do not rename tables, columns, constraints or error codes.
  2. Only the Spring Boot API service connects to the database. Workers never do: they send results back through the job-results queue.
  3. Every request runs inside one transaction that first sets the session values used by row-level security (section 7). This works with Neon's pooled connection because SET LOCAL lasts only for the transaction.
  4. Value lists are text columns with CHECK constraints, not PostgreSQL enums, so they can change with a simple migration. Java enums must match them exactly.
  5. All times are timestamptz in UTC. "Today" and "this week" are calculated in Asia/Kolkata.
  6. IDs are random UUIDs (gen_random_uuid()). Records created offline by the app (attempts, answers, reading tests) use a UUID generated on the device, which also makes repeated uploads harmless.
  7. Files live in Cloudflare R2. Tables store only the object key. All file access uses signed URLs valid for 10 minutes.

How shared lessons work: a parent's upload creates family_chapters (private). The admin links each one to a catalog_chapter (a prepared chapter, either new or already made for another family with the same book). Lessons and questions belong to the catalog chapter, so a second family with the same book gets them at once. Families never see each other's files.

4Tables (SQL)

All objects live in schema app. Every table with updated_at gets the set_updated_at trigger (section 6).

4.1 Reference and people

CREATE SCHEMA app;

CREATE TABLE app.subjects (
  id         smallint PRIMARY KEY,
  code       text NOT NULL UNIQUE CHECK (code IN ('english','hindi','maths','evs','computer')),
  name       text NOT NULL,
  active     boolean NOT NULL DEFAULT true
);

CREATE TABLE app.class_subjects (
  class_level smallint NOT NULL CHECK (class_level BETWEEN 1 AND 5),
  subject_id  smallint NOT NULL REFERENCES app.subjects(id),
  PRIMARY KEY (class_level, subject_id)
);

CREATE TABLE app.families (
  id                    uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  keycloak_user_id      text NOT NULL UNIQUE,
  consent_version       text NOT NULL,
  consent_at            timestamptz NOT NULL,
  pin_hash              text,                                   -- BCrypt (strength 12); NULL until set
  pin_failed_count      smallint NOT NULL DEFAULT 0,
  pin_locked_until      timestamptz,
  daily_limit_minutes   smallint NOT NULL DEFAULT 30 CHECK (daily_limit_minutes IN (20,30,45,60)),
  sound_enabled         boolean NOT NULL DEFAULT true,
  celebrations_enabled  boolean NOT NULL DEFAULT true,
  deletion_requested_at timestamptz,
  created_at            timestamptz NOT NULL DEFAULT now(),
  updated_at            timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE app.children (
  id          uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id   uuid NOT NULL REFERENCES app.families(id) ON DELETE CASCADE,
  first_name  varchar(40) NOT NULL CHECK (length(btrim(first_name)) > 0),
  age         smallint NOT NULL CHECK (age BETWEEN 4 AND 12),
  class_level smallint NOT NULL CHECK (class_level BETWEEN 1 AND 5),
  avatar_key  text NOT NULL CHECK (avatar_key IN ('cat','dog','lion','panda','rabbit','tiger',
                                                  'fox','koala','monkey','penguin','frog','unicorn')),
  stars_total integer NOT NULL DEFAULT 0 CHECK (stars_total >= 0),
  created_at  timestamptz NOT NULL DEFAULT now(),
  updated_at  timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX children_family_idx ON app.children(family_id);

CREATE TABLE app.staff_users (
  id               uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  keycloak_user_id text NOT NULL UNIQUE,
  display_name     varchar(80) NOT NULL,
  role             text NOT NULL CHECK (role IN ('content_admin','super_admin')),
  active           boolean NOT NULL DEFAULT true,
  created_at       timestamptz NOT NULL DEFAULT now(),
  updated_at       timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE app.catalog_chapters (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  catalog_book_id uuid NOT NULL REFERENCES app.catalog_books(id) ON DELETE RESTRICT,
  number          smallint NOT NULL CHECK (number > 0),

  created_at      timestamptz NOT NULL DEFAULT now(),
  updated_at      timestamptz NOT NULL DEFAULT now(),
  UNIQUE (catalog_book_id, number)
);

CREATE TABLE app.content_items (
  id                   uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  kind                 text NOT NULL CHECK (kind IN ('lesson','question','passage')),
  scope                text NOT NULL CHECK (scope IN ('library','catalog_chapter')),
  class_level          smallint NOT NULL CHECK (class_level BETWEEN 1 AND 5),
  subject_id           smallint NOT NULL REFERENCES app.subjects(id),
  topic_id             uuid REFERENCES app.topics(id),
  catalog_chapter_id   uuid REFERENCES app.catalog_chapters(id) ON DELETE RESTRICT,
  question_type        text CHECK (question_type IN ('mcq','true_false','fill_blank','match')),
  difficulty           smallint NOT NULL DEFAULT 2 CHECK (difficulty BETWEEN 1 AND 3),
  lang                 text NOT NULL CHECK (lang IN ('en','hi')),
  status               text NOT NULL DEFAULT 'draft' CHECK (status IN ('draft','in_check','published','archived')),
  current_version_id   uuid,
  published_version_id uuid,
  published_at         timestamptz,
  created_by           uuid NOT NULL REFERENCES app.staff_users(id),
  created_at           timestamptz NOT NULL DEFAULT now(),
  updated_at           timestamptz NOT NULL DEFAULT now(),
  archived_at          timestamptz,
  CONSTRAINT content_scope_ck CHECK (
    (scope = 'library' AND catalog_chapter_id IS NULL AND topic_id IS NOT NULL) OR
    (scope = 'catalog_chapter' AND catalog_chapter_id IS NOT NULL)),
  CONSTRAINT content_qtype_ck CHECK ((kind = 'question') = (question_type IS NOT NULL)),
  CONSTRAINT content_passage_ck CHECK (kind <> 'passage' OR scope = 'library')
);
CREATE INDEX content_items_lookup_idx  ON app.content_items(kind, status, class_level, subject_id, topic_id);
CREATE INDEX content_items_chapter_idx ON app.content_items(catalog_chapter_id, kind, status);

CREATE TABLE app.content_versions (
  id                  uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  item_id             uuid NOT NULL REFERENCES app.content_items(id) ON DELETE CASCADE,
  version_no          integer NOT NULL,
  title               varchar(200) NOT NULL,
  body                jsonb NOT NULL,                 -- format per kind: section 5
  solution            jsonb,                          -- questions only
  asset_keys          text[] NOT NULL DEFAULT '{}',   -- R2 keys of images used
  narration_audio_key text,                           -- lessons: generated narration
  duration_seconds    integer CHECK (duration_seconds >= 0),
  created_by          uuid NOT NULL REFERENCES app.staff_users(id),
  created_at          timestamptz NOT NULL DEFAULT now(),
  UNIQUE (item_id, version_no)
);
ALTER TABLE app.content_items
  ADD CONSTRAINT content_current_fk   FOREIGN KEY (current_version_id)   REFERENCES app.content_versions(id) DEFERRABLE INITIALLY DEFERRED,
  ADD CONSTRAINT content_published_fk FOREIGN KEY (published_version_id) REFERENCES app.content_versions(id) DEFERRABLE INITIALLY DEFERRED;

CREATE TABLE app.content_checks (
  id             uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  version_id     uuid NOT NULL REFERENCES app.content_versions(id) ON DELETE CASCADE,
  status         text NOT NULL DEFAULT 'running' CHECK (status IN ('running','passed','failed','error')),
  model          text NOT NULL,
  critical_count integer NOT NULL DEFAULT 0,
  warning_count  integer NOT NULL DEFAULT 0,
  error_message  text,
  started_at     timestamptz NOT NULL DEFAULT now(),
  finished_at    timestamptz
);
CREATE INDEX content_checks_version_idx ON app.content_checks(version_id, started_at DESC);

CREATE TABLE app.content_check_flags (
  id              uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  check_id        uuid NOT NULL REFERENCES app.content_checks(id) ON DELETE CASCADE,
  severity        text NOT NULL CHECK (severity IN ('critical','warning')),
  category        text NOT NULL CHECK (category IN ('age_suitability','fact_mismatch','wrong_answer',
                                                   'language','difficulty','bias','personal_info','other')),
  location        jsonb NOT NULL,                 -- {"path": "scenes[2].text"} or {"field": "solution"}
  reason          text NOT NULL,
  resolution      text NOT NULL DEFAULT 'open' CHECK (resolution IN ('open','fixed','accepted')),
  resolution_note varchar(500),
  resolved_by     uuid REFERENCES app.staff_users(id),
  resolved_at     timestamptz,
  CONSTRAINT flag_accept_warning_only CHECK (resolution <> 'accepted' OR severity = 'warning'),
  CONSTRAINT flag_accept_needs_note   CHECK (resolution <> 'accepted' OR resolution_note IS NOT NULL)
);

  number             smallint NOT NULL CHECK (number > 0),
  title              varchar(200) NOT NULL,
  catalog_chapter_id uuid REFERENCES app.catalog_chapters(id),
  status             text NOT NULL DEFAULT 'processing' CHECK (status IN

  page_count         smallint NOT NULL DEFAULT 0 CHECK (page_count BETWEEN 0 AND 300),
  created_at         timestamptz NOT NULL DEFAULT now(),

  CONSTRAINT family_chapter_number_uq UNIQUE (family_book_id, number) DEFERRABLE INITIALLY DEFERRED

CREATE TABLE app.uploads (
  id                uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  family_id         uuid NOT NULL REFERENCES app.families(id) ON DELETE CASCADE,
  family_book_id    uuid NOT NULL REFERENCES app.family_books(id) ON DELETE CASCADE,
  family_chapter_id uuid REFERENCES app.family_chapters(id) ON DELETE CASCADE,  -- NULL for a whole book
  source            text NOT NULL CHECK (source IN ('pdf','photos')),
  status            text NOT NULL DEFAULT 'awaiting_files' CHECK (status IN
                      ('awaiting_files','scanning','rejected','processing','processed','failed')),
  file_count        smallint NOT NULL CHECK (file_count BETWEEN 1 AND 50),
  reject_reason     text CHECK (reject_reason IN ('infected','wrong_type','too_large','too_many_pages','unreadable')),
  created_at        timestamptz NOT NULL DEFAULT now(),
  updated_at        timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT upload_pdf_single CHECK (source <> 'pdf' OR file_count = 1)
  -- V13 (Plan 5.3): retake_page_id uuid REFERENCES app.pages(id) ON DELETE CASCADE,
  --   CONSTRAINT upload_retake_one_photo CHECK (retake_page_id IS NULL OR (source = 'photos' AND file_count = 1))
);

CREATE TABLE app.upload_files (
  id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  upload_id    uuid NOT NULL REFERENCES app.uploads(id) ON DELETE CASCADE,
  family_id    uuid NOT NULL REFERENCES app.families(id) ON DELETE CASCADE,
  object_key   text NOT NULL UNIQUE,
  content_type text NOT NULL CHECK (content_type IN ('application/pdf','image/jpeg','image/png')),
  byte_size    bigint NOT NULL CHECK (
                 (content_type = 'application/pdf' AND byte_size BETWEEN 1 AND 52428800) OR
                 (content_type <> 'application/pdf' AND byte_size BETWEEN 1 AND 10485760)),
  scan_status  text NOT NULL DEFAULT 'pending' CHECK (scan_status IN ('pending','clean','infected')),
  -- V13 (Plan 5.3): position smallint CHECK (position >= 0), the parent's file order
  created_at   timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE app.pages (
  id                  uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  family_chapter_id   uuid NOT NULL REFERENCES app.family_chapters(id) ON DELETE CASCADE,
  family_id           uuid NOT NULL REFERENCES app.families(id) ON DELETE CASCADE,
  upload_file_id      uuid NOT NULL REFERENCES app.upload_files(id) ON DELETE CASCADE,

  image_key           text NOT NULL,
  ocr_text            text,
  ocr_confidence      numeric(5,2) CHECK (ocr_confidence BETWEEN 0 AND 100),
  quality             text NOT NULL DEFAULT 'ok' CHECK (quality IN ('ok','unclear','retaken')),
  narration_audio_key text,
  word_timings        jsonb,          -- [{"w":"bird","start_ms":1200,"end_ms":1480}, ...]

  updated_at          timestamptz NOT NULL DEFAULT now(),

CREATE TABLE app.prep_tasks (

Implemented in Plan 4.2 (decided 21 Oct 2026)

  • The schema files are lesson.body, passage.body and question-<type>.body / .solution (*.schema.json, draft 2020-12). They are checked by a small in-house validator (common/jsonschema; no extra dependency) that supports type, required, properties, additionalProperties, items, enum, const, minimum/maximum, minLength/maxLength, minItems/maxItems and pattern, and refuses any other keyword when it loads.
  • Errors are 400 VALIDATION_FAILED with one field code per path, e.g. "body.scenes[2].text": "REQUIRED", "solution.correct[0]": "INVALID". Codes: REQUIRED (missing, null or blank), INVALID (wrong type, value or rule), INVALID_LENGTH, OUT_OF_RANGE, INVALID_FORMAT.
  • Lesson: 1–40 scenes. id matches ^[a-z0-9_-]{1,20}$ and is unique; duration_ms 0–60000. Each scene has exactly one of text (1–500 characters, narration) or quiz (a published question's id). visual is null or {illustration, animation}: illustration is a vendored key from Guidelines §9 (the list is kept equal to libs/ui/assets/manifest.json by scripts/check-assets.mjs), animation is one of fade-up, slide-in, pop, none (reduced motion shows none). duration_seconds is the scenes' total, rounded up.
  • Questions: prompt 1–300 characters, explanation 1–500 (required). mcq: 2–4 options with ids a–d (answer buttons A–D), exactly one correct id among them. fill_blank: the prompt contains ___, 2–4 different choices, correct is one of them. match: 2–5 items per side with ids ^[a-z0-9]{1,4}$, one pair per left item, every id used once. Lessons and passages have no solution.
  • Passage: text 1–5000 characters, word_count must equal the number of space-separated words in text (the editor fills it in), level 1 (easy) to 3 (hard).

CREATE FUNCTION app.track_mistakes() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NOT NEW.is_correct THEN
    INSERT INTO app.mistakes (child_id, question_item_id, family_id, topic_id, last_wrong_at)
    VALUES (NEW.child_id, NEW.question_item_id, NEW.family_id, NEW.topic_id, NEW.answered_at)
    ON CONFLICT (child_id, question_item_id) DO UPDATE
      SET wrong_count = app.mistakes.wrong_count + 1, correct_streak = 0,
          last_wrong_at = EXCLUDED.last_wrong_at, resolved_at = NULL;
  ELSE
    UPDATE app.mistakes
       SET correct_streak = correct_streak + 1,
           resolved_at = CASE WHEN correct_streak + 1 >= 2 THEN NEW.answered_at ELSE NULL END
     WHERE child_id = NEW.child_id AND question_item_id = NEW.question_item_id AND resolved_at IS NULL;
  END IF;
  RETURN NULL;
END $$;

CREATE FUNCTION app.move_child_to_next_class(p_child uuid) RETURNS smallint LANGUAGE plpgsql AS $$
DECLARE v_class smallint;
BEGIN
  SELECT class_level INTO v_class FROM app.children WHERE id = p_child FOR UPDATE;
  IF v_class IS NULL THEN RAISE EXCEPTION 'NOT_FOUND' USING ERRCODE = 'P0002'; END IF;
  IF v_class >= 5 THEN RAISE EXCEPTION 'MAX_CLASS' USING ERRCODE = 'P0001'; END IF;
  DELETE FROM app.activity_attempts WHERE child_id = p_child;   -- cascades answers, reading tests

  DELETE FROM app.child_books       WHERE child_id = p_child;
  UPDATE app.children SET class_level = v_class + 1, stars_total = 0 WHERE id = p_child;
  RETURN v_class + 1;
END $$;

CREATE FUNCTION app.child_week_summary(p_child uuid, p_week_start date)
RETURNS TABLE (minutes integer, active_days integer, chapters_done integer, worksheets_done integer, areas jsonb)
LANGUAGE sql STABLE AS $$
  WITH w AS (SELECT (p_week_start::timestamp AT TIME ZONE 'Asia/Kolkata') AS f,
                    ((p_week_start + 7)::timestamp AT TIME ZONE 'Asia/Kolkata') AS t)
  SELECT
    (SELECT count(*)::int FROM app.usage_minutes u, w WHERE u.child_id = p_child AND u.minute_at >= w.f AND u.minute_at < w.t),
    (SELECT count(DISTINCT (u.minute_at AT TIME ZONE 'Asia/Kolkata')::date)::int FROM app.usage_minutes u, w
      WHERE u.child_id = p_child AND u.minute_at >= w.f AND u.minute_at < w.t),
    (SELECT count(DISTINCT a.family_chapter_id)::int FROM app.activity_attempts a, w
      WHERE a.child_id = p_child AND a.kind = 'lesson' AND a.completed_at >= w.f AND a.completed_at < w.t),
    (SELECT count(*)::int FROM app.worksheets ws, w WHERE ws.child_id = p_child AND ws.created_at >= w.f AND ws.created_at < w.t),
    (SELECT COALESCE(jsonb_agg(x ORDER BY x.score DESC), '[]') FROM (
        SELECT t.id AS topic_id, t.name, s.code AS subject,
               COALESCE(m.open_mistakes,0) * 1 + COALESCE(ws.ticks,0) * 2 AS score
          FROM app.topics t JOIN app.subjects s ON s.id = t.subject_id
          LEFT JOIN (SELECT topic_id, count(*) AS open_mistakes FROM app.mistakes
                      WHERE child_id = p_child AND resolved_at IS NULL GROUP BY topic_id) m ON m.topic_id = t.id
          LEFT JOIN (SELECT topic_id, count(*) AS ticks FROM app.worksheet_weak_spots
                      WHERE child_id = p_child AND marked_at > now() - interval '30 days' GROUP BY topic_id) ws ON ws.topic_id = t.id
         WHERE COALESCE(m.open_mistakes,0) + COALESCE(ws.ticks,0) > 0
         ORDER BY score DESC LIMIT 3) x);
$$;

CREATE FUNCTION app.complete_family_deletion(p_family uuid, p_staff uuid) RETURNS void LANGUAGE plpgsql AS $$

CREATE FUNCTION app.run_retention() RETURNS TABLE (object_key text) LANGUAGE plpgsql AS $$
BEGIN
  RETURN QUERY
  WITH rec AS (
         SELECT id, recording_key FROM app.reading_tests
          WHERE recording_key IS NOT NULL AND recording_expires_at < now() FOR UPDATE),
       rec_clear AS (
         UPDATE app.reading_tests r SET recording_key = NULL
           FROM rec WHERE r.id = rec.id RETURNING rec.recording_key AS k),
       pdf AS (
         SELECT id, pdf_key FROM app.worksheets
          WHERE pdf_key IS NOT NULL AND pdf_expires_at < now() FOR UPDATE),
       pdf_clear AS (
         UPDATE app.worksheets w SET pdf_key = NULL, pdf_expires_at = NULL
           FROM pdf WHERE w.id = pdf.id RETURNING pdf.pdf_key AS k)
  SELECT k FROM rec_clear UNION ALL SELECT k FROM pdf_clear;

  DELETE FROM app.messages       WHERE expires_at   < now();
  DELETE FROM app.audit_log      WHERE created_at   < now() - interval '1 year';
  DELETE FROM app.usage_events   WHERE occurred_at  < now() - interval '1 year';
  DELETE FROM app.usage_minutes  WHERE minute_at    < now() - interval '400 days';
  DELETE FROM app.processed_jobs WHERE processed_at < now() - interval '14 days';
END $$;

SET LOCAL app.actor     = 'parent' | 'staff';
